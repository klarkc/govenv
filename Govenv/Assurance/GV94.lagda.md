# GV94 assurance

GV94 is statically satisfied by rejecting an active phase with no pending governance, accepting an active phase with pending work, and deriving README current/next-work state from the typed roadmap rather than an independent progress summary.

```agda
{-# OPTIONS --safe #-}

module Govenv.Assurance.GV94 where

open import Agda.Builtin.Bool using (true; false)
open import Agda.Builtin.Equality using (_≡_; refl)
open import Agda.Builtin.List using ([]; _∷_)
open import Govenv.Kernel.Assurance using (StaticEvidence; staticEvidence)
open import Govenv.Kernel.Identifier using (P; GV; someIdentifier)
open import Govenv.Kernel.Roadmap using
  (Roadmap; roadmapOf; roadmapIntegrity; _▣; _◇; _✓; _├_)
open import Govenv.Materialization.Readme using (Current; activeCurrent; currentOf)

sampleRoadmap : Roadmap
sampleRoadmap = roadmapOf
  ((P 1 "active" ▣) (GV 942 "pending" ◇))

record Proposition : Set where
  constructor satisfied
  field
    staleActivePhaseRejected :
      roadmapIntegrity ((P 1 "active" ▣) (GV 940 "terminal" ✓)) ≡ false
    activePhaseWithWorkAccepted :
      roadmapIntegrity
        ((P 1 "active" ▣)
          ((GV 941 "terminal" ✓) ├ (GV 942 "pending" ◇))) ≡ true
    currentProjectionDerived :
      currentOf sampleRoadmap ≡
        activeCurrent "Current"
          (someIdentifier (P 1 "active"))
          (someIdentifier (GV 942 "pending") ∷ [])

proof : Proposition
proof = satisfied refl refl refl

evidence : StaticEvidence 94
evidence = staticEvidence Proposition proof
```
